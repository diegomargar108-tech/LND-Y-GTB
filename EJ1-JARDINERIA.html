1. SELECT CodigoOficina , Ciudad FROM Oficinas;
  2. SELECT COUNT(*) AS TotalEmpleados FROM Empleados;
  3. SELECT Pais, COUNT(*) AS CantidadClientes FROM Clientes GROUP BY Pais;
  4. SELECT AVG(Cantidad) AS PagoMedio FROM Pagos WHERE YEAR(FechaPago) = 2006;
  5. SELECT Estado , COUNT(*) AS NumPedidos FROM Pedidos GROUP BY Estado ORDER BY NumPedidos ASC;
  6. SELECT MAX(PrecioVenta) AS PrecioMasCaro , MIN(PrecioVenta) AS PrecioMasBajo FROM Productos;
  7. SELECT Ciudad, Telefono FROM Oficinas WHERE Pais IN ('EEUU', 'USA', 'Estados Unidos');
  8. SELECT Nombre, Apellido1, Apellido2, Email FROM Empleados WHERE CodigoJefe = (SELECT CodigoEmpleado FROM Empleados WHERE Nombre = 'Alberto' AND Apellido1 = 'Soria');
  9. SELECT Puesto, Nombre, Apellido1, Apellido2, Email FROM Empleados WHERE CodigoJefe IS NULL;
  10. SELECT Nombre, Apellido1, Apellido2, Puesto FROM Empleados WHERE Puesto <> 'Representante Ventas';
  11. SELECT COUNT(*) AS TotalClientes FROM Clientes;
  12. SELECT NombreCliente FROM Clientes WHERE Pais IN ('Spain', 'España');
  13. SELECT Pais, COUNT(*) AS NumeroClientes FROM Clientes GROUP BY Pais;
  14. SELECT COUNT(*) AS ClientesMadrid FROM Clientes WHERE Ciudad = 'Madrid';
  15. SELECT COUNT(*) AS ClientesCiudadesM FROM Clientes WHERE Ciudad LIKE 'M%';
  16. SELECT CodigoEmpleadoRepVentas, COUNT(*) AS NumeroClientes FROM Clientes WHERE CodigoEmpleadoRepVentas IS NOT NULL GROUP BY CodigoEmpleadoRepVentas;
  17. SELECT COUNT(*) AS ClientesSinRep FROM Clientes WHERE CodigoEmpleadoRepVentas IS NULL;
  18. SELECT MIN(FechaPago) AS PrimerPago, MAX(FechaPago) AS UltimoPago FROM Pagos;
  19. SELECT DISTINCT CodigoCliente FROM Pagos WHERE FechaPago BETWEEN '2008-01-01' AND '2008-12-31';
  20. SELECT DISTINCT Estado FROM Pedidos;
  21. SELECT CodigoPedido, CodigoCliente, FechaEsperada, FechaEntrega FROM Pedidos WHERE FechaEntrega > FechaEsperada;
  22. SELECT CodigoPedido, COUNT(DISTINCT CodigoProducto) AS CantidadProductos FROM DetallePedidos GROUP BY CodigoPedido;
  23. SELECT CodigoProducto, SUM(Cantidad) AS TotalCantidad FROM DetallePedidos GROUP BY CodigoProducto ORDER BY TotalCantidad DESC LIMIT 20;
  24. SELECT CodigoPedido, CodigoCliente, FechaEsperada, FechaEntrega FROM Pedidos WHERE FechaEntrega <= ADDDATE(FechaEsperada, INTERVAL -2 DAY);
  25. SELECT SUM(Cantidad * PrecioUnidad) AS BaseImponible, SUM(Cantidad * PrecioUnidad) * 0.18 AS IVA, SUM(Cantidad * PrecioUnidad) * 1.18 AS TotalFacturado FROM DetallePedidos;
  26. SELECT CodigoProducto, SUM(Cantidad * PrecioUnidad) AS BaseImponible, SUM(Cantidad * PrecioUnidad) * 0.18 AS IVA, SUM(Cantidad * PrecioUnidad) * 1.18 AS TotalFacturado FROM DetallePedidos WHERE CodigoProducto LIKE 'FR%' GROUP BY CodigoProducto;
